### Open Inner Circle: decisions.
###
### One browser for every German advisor, general and admiral: Next and Previous step
### through the list, Browse changes who it lists, and the action decisions work on the
### character it points at (the country variable oic_candidate). Each decision does its
### work in a scripted effect inside hidden_effect, next to a readable custom tooltip.
### The AI never takes these.

oic_category = {

	oic_make_leader = {
		icon = oic_leader
		priority = 6

		visible = {
			has_variable = oic_candidate
			var:oic_candidate = {
				is_country_leader = no
			}
		}

		available = {
			oic_has_candidate = yes
		}

		cost = 100

		complete_effect = {
			custom_effect_tooltip = oic_make_leader_tt
			hidden_effect = {
				oic_make_candidate_leader = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	oic_seat_in_inner_circle = {
		icon = oic_seat
		priority = 5

		visible = {
			has_variable = oic_candidate
			var:oic_candidate = {
				is_country_leader = no
				NOT = { has_character_flag = oic_member }
			}
		}

		available = {
			has_variable = oic_candidate
			custom_trigger_tooltip = {
				tooltip = oic_candidate_is_advisor_or_commander_tt
				var:oic_candidate = {
					oic_is_advisor_or_commander = yes
				}
			}
		}

		cost = 25

		complete_effect = {
			custom_effect_tooltip = oic_seat_tt
			hidden_effect = {
				oic_seat_candidate = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	oic_dismiss_from_inner_circle = {
		icon = oic_dismiss
		priority = 5

		visible = {
			has_variable = oic_candidate
			var:oic_candidate = {
				has_character_flag = oic_member
			}
		}

		complete_effect = {
			custom_effect_tooltip = oic_dismiss_tt
			hidden_effect = {
				oic_dismiss_candidate = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	oic_next_candidate = {
		icon = oic_next
		priority = 4

		complete_effect = {
			custom_effect_tooltip = oic_next_candidate_tt
			hidden_effect = {
				set_temp_variable = { oic_step = 1 }
				oic_move_cursor = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	oic_previous_candidate = {
		icon = oic_previous
		priority = 3

		complete_effect = {
			custom_effect_tooltip = oic_previous_candidate_tt
			hidden_effect = {
				set_temp_variable = { oic_step = -1 }
				oic_move_cursor = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}

	oic_change_filter = {
		icon = oic_filter
		priority = 2

		complete_effect = {
			custom_effect_tooltip = oic_change_filter_tt
			hidden_effect = {
				oic_cycle_filter = yes
			}
		}

		ai_will_do = {
			factor = 0
		}
	}
}
