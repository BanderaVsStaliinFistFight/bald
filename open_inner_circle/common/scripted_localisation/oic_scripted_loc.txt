### Open Inner Circle: scripted localisation for the browser. Country scope (Germany).

defined_text = {
	name = oic_GetCandidateLine
	text = {
		trigger = { oic_has_candidate = yes }
		localization_key = oic_candidate_line
	}
	text = {
		localization_key = oic_no_candidate_line
	}
}

defined_text = {
	name = oic_GetFilterName
	text = {
		trigger = { check_variable = { oic_filter = 1 } }
		localization_key = oic_filter_advisors
	}
	text = {
		trigger = { check_variable = { oic_filter = 2 } }
		localization_key = oic_filter_generals
	}
	text = {
		trigger = { check_variable = { oic_filter = 3 } }
		localization_key = oic_filter_admirals
	}
	text = {
		trigger = { check_variable = { oic_filter = 4 } }
		localization_key = oic_filter_members
	}
	text = {
		localization_key = oic_filter_everyone
	}
}

defined_text = {
	name = oic_GetCandidateRole
	text = {
		trigger = {
			var:oic_candidate = {
				is_advisor = yes
				is_unit_leader = yes
			}
		}
		localization_key = oic_role_advisor_and_commander
	}
	text = {
		trigger = { var:oic_candidate = { is_political_advisor = yes } }
		localization_key = oic_role_political_advisor
	}
	text = {
		trigger = { var:oic_candidate = { is_advisor = yes } }
		localization_key = oic_role_military_advisor
	}
	text = {
		trigger = { var:oic_candidate = { is_field_marshal = yes } }
		localization_key = oic_role_field_marshal
	}
	text = {
		trigger = { var:oic_candidate = { is_army_leader = yes } }
		localization_key = oic_role_general
	}
	text = {
		trigger = { var:oic_candidate = { is_navy_leader = yes } }
		localization_key = oic_role_admiral
	}
	text = {
		localization_key = oic_role_former_leader
	}
}

defined_text = {
	name = oic_GetCandidateStatus
	text = {
		trigger = { var:oic_candidate = { is_country_leader = yes } }
		localization_key = oic_status_leader
	}
	text = {
		trigger = { var:oic_candidate = { has_character_flag = oic_member } }
		localization_key = oic_status_member
	}
	text = {
		localization_key = oic_status_outsider
	}
}
